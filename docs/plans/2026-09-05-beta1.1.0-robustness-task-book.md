def git_output(*args):
    return subprocess.check_output(["git", "-C", str(repo), *args]).decode().strip()

index_path = Path(git_output("rev-parse", "--git-path", "index"))
if not index_path.is_absolute():
    index_path = repo / index_path
ledger_path = ledger.ledger_file(repo, "memory")
before_head = git_output("rev-parse", "HEAD")
before_index = index_path.read_bytes()
before_ledger_bytes = ledger_path.read_bytes()
real_run = subprocess.run

def reject_commit(command, **kwargs):
    if command[:3] == ["git", "-C", str(repo)] and command[3] == "commit":
        return subprocess.CompletedProcess(command, 1, stdout="", stderr="fixture commit rejected")
    return real_run(command, **kwargs)

with mock.patch.object(ledger.subprocess, "run", side_effect=reject_commit):
    result = ledger.domain_ledger_write(
        kind="memory", operation="update", topic="fixture", content="revision two",
        actor="shiguan", authority="autonomous", write_set=["memory"], root=repo,
    )

assert git_output("rev-parse", "HEAD") == before_head
assert index_path.read_bytes() == before_index
assert ledger_path.read_bytes() == before_ledger_bytes
